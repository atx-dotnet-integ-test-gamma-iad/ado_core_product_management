KEY SQL CONVERSIONS APPLIED

Pattern 1: INSERT with Identity Capture
  FROM: DECLARE @Id INT; INSERT...; SET @Id = SCOPE_IDENTITY(); SELECT @Id
  TO:   WITH inserted AS (INSERT ... RETURNING Id) SELECT Id FROM inserted

Pattern 2: UPDATE with Old Value Capture
  FROM: DECLARE @Old; SELECT @Old = Value FROM Table; UPDATE...; 
  TO:   WITH old_values AS (SELECT Value), product_update AS (UPDATE...)

Pattern 3: DELETE with Value Preservation
  FROM: DECLARE @Old; SELECT @Old = Value FROM Table; DELETE...
  TO:   WITH old_values AS (SELECT Value), product_delete AS (DELETE...)

Pattern 4: Date Functions
  FROM: GETDATE()
  TO:   CURRENT_TIMESTAMP

Pattern 5: Transaction Syntax
  FROM: BEGIN TRANSACTION; ... COMMIT;
  TO:   Application-level: BeginTransactionAsync(), CommitAsync()
