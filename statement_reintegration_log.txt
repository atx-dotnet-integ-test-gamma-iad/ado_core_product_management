        COUNT(*) OVER() as TotalProducts
    FROM Products
)

    p.Name,
    p.Description,
    p.Price,
    p.StockQuantity,
    p.CreatedDate,

WITH ProductHistory AS (
    SELECT 
        ProductId,
        LAG(Price) OVER (ORDER BY ModifiedDate) as PreviousPrice,
        LAG(StockQuantity) OVER (ORDER BY ModifiedDate) as PreviousStock
    FROM Products
    WHERE ProductId = @ProductId
)
SELECT 
    p.ProductId,
    p.Name,
    p.Description,
    p.Price,
    p.StockQuantity,
    p.CreatedDate,
    p.ModifiedDate,
    ph.PreviousPrice,
    ph.PreviousStock,
    CASE 
        WHEN ph.PreviousPrice IS NOT NULL THEN 
            ROUND(((p.Price - ph.PreviousPrice) / ph.PreviousPrice) * 100, 2)
        ELSE NULL
    END as PriceChangePercentage
FROM Products p
LEFT JOIN ProductHistory ph ON p.ProductId = ph.ProductId
WHERE p.ProductId = @ProductId

STATEMENT 3: InsertProductAsync - Transaction with SCOPE_IDENTITY
===============================================================================
Source Location: ProductRepository.cs, line ~116-155
Status: MAJOR CHANGES APPLIED
Reason: SCOPE_IDENTITY() and GETDATE() are SQL Server specific

BEFORE (SQL Server):
DECLARE @NewProductId INT;

BEGIN TRANSACTION;
    -- Insert the new product
    INSERT INTO Products (Name, Description, Price, StockQuantity)
    VALUES (@Name, @Description, @Price, @StockQuantity);
    
    SET @NewProductId = SCOPE_IDENTITY();
    
    -- Log the insertion
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    VALUES (@NewProductId, 'INSERT', NULL, @Price, NULL, @StockQuantity, GETDATE());
    
    -- Update product statistics
    UPDATE ProductStats
    SET 
        TotalProducts = TotalProducts + 1,
        AveragePrice = (AveragePrice * TotalProducts + @Price) / (TotalProducts + 1),
        LastUpdated = GETDATE()
    WHERE StatId = 1;
COMMIT;

SELECT @NewProductId;

AFTER (PostgreSQL):
WITH inserted_product AS (
    INSERT INTO Products (Name, Description, Price, StockQuantity)
    VALUES (@Name, @Description, @Price, @StockQuantity)
    RETURNING ProductId
),
inserted_history AS (
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    SELECT ProductId, 'INSERT', NULL, @Price, NULL, @StockQuantity, CURRENT_TIMESTAMP
    FROM inserted_product
    RETURNING ProductId
)
UPDATE ProductStats
SET 
    TotalProducts = TotalProducts + 1,
    AveragePrice = (AveragePrice * TotalProducts + @Price) / (TotalProducts + 1),
    LastUpdated = CURRENT_TIMESTAMP
WHERE StatId = 1
RETURNING (SELECT ProductId FROM inserted_product);

CHANGES APPLIED:
1. Removed DECLARE @NewProductId INT - PostgreSQL uses RETURNING instead
2. Removed BEGIN TRANSACTION/COMMIT - Transaction managed at application level
3. Replaced SCOPE_IDENTITY() with RETURNING clause in first INSERT
4. Replaced GETDATE() with CURRENT_TIMESTAMP (2 occurrences)
5. Used CTEs to chain operations and pass ProductId between statements
6. Changed SELECT @NewProductId to RETURNING (SELECT ProductId FROM inserted_product)

NOTES:
- PostgreSQL RETURNING clause captures inserted ID immediately
- CTEs with data-modifying statements allow operation chaining
- CURRENT_TIMESTAMP is standard SQL, GETDATE() is SQL Server specific
- Transaction control moved to C# code using NpgsqlConnection.BeginTransaction()

WITH old_values AS (
    SELECT Price as OldPrice, StockQuantity as OldStock
    FROM Products
    WHERE ProductId = @ProductId
),
updated_product AS (
    UPDATE Products
    SET 
        Name = @Name,
        Description = @Description,
        Price = @Price,
        StockQuantity = @StockQuantity,
        ModifiedDate = CURRENT_TIMESTAMP
    WHERE ProductId = @ProductId
    RETURNING ProductId
),
inserted_history AS (
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    SELECT @ProductId, 'UPDATE', ov.OldPrice, @Price, ov.OldStock, @StockQuantity, CURRENT_TIMESTAMP
    FROM old_values ov
    RETURNING ProductId
)
UPDATE ProductStats
SET 
    AveragePrice = (AveragePrice * TotalProducts - (SELECT OldPrice FROM old_values) + @Price) / TotalProducts,
    LastUpdated = CURRENT_TIMESTAMP
WHERE StatId = 1;

2. Removed DECLARE statements for @OldPrice and @OldStock
3. Created old_values CTE to capture old values using SELECT
4. Replaced GETDATE() with CURRENT_TIMESTAMP (3 occurrences)
5. Used CTEs to chain UPDATE, INSERT, and second UPDATE operations
6. Reference old values from CTE using (SELECT OldPrice FROM old_values)

NOTES:
- PostgreSQL doesn't support variable declarations in SQL statements
- CTEs provide elegant alternative for capturing and passing values
- All operations now use standard SQL features
- RETURNING clause allows operation chaining

WITH StockAnalysis AS (
    SELECT 
        p.*,
        AVG(StockQuantity) OVER() as AvgStock,
        MIN(StockQuantity) OVER() as MinStock,

)

    sa.*,
    CASE 
        WHEN StockQuantity <= @Threshold THEN 'Critical'
        WHEN StockQuantity <= AvgStock * 0.5 THEN 'Low'
        ELSE 'Adequate'
    END as StockStatus,
    ROUND((StockQuantity / AvgStock) * 100, 2) as StockPercentageOfAverage
FROM StockAnalysis sa
WHERE StockQuantity <= @Threshold
ORDER BY StockQuantity
