ORIGINAL SQL (SQL Server Syntax - INVALID for PostgreSQL):

DECLARE @NewProductId INT;

-- PostgreSQL transaction (use BEGIN or application-level transaction)
    -- Insert the new product
    INSERT INTO Products (Name, Description, Price, StockQuantity)
    VALUES (@Name, @Description, @Price, @StockQuantity);
    
    -- Use RETURNING clause in INSERT statement
    
    -- Log the insertion

    VALUES (@NewProductId, 'INSERT', NULL, @Price, NULL, @StockQuantity, CURRENT_TIMESTAMP);
    
    -- Update product statistics

    SET

-- PostgreSQL COMMIT

SELECT @NewProductId;

PROBLEMS:
- DECLARE @NewProductId INT - Not supported in PostgreSQL
- VALUES (@NewProductId, ...) - Variable not set
- SELECT @NewProductId - Variable syntax not supported

FIXED SQL (PostgreSQL CTE with RETURNING):
------------------------------------------
WITH inserted_product AS (
    INSERT INTO Products (Name, Description, Price, StockQuantity)
    VALUES (@Name, @Description, @Price, @StockQuantity)
    RETURNING ProductId
),
history_insert AS (
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    SELECT ProductId, 'INSERT', NULL, @Price, NULL, @StockQuantity, CURRENT_TIMESTAMP
    FROM inserted_product
    RETURNING ProductId
)
UPDATE ProductStats
SET 
    TotalProducts = TotalProducts + 1,
    AveragePrice = (AveragePrice * TotalProducts + @Price) / (TotalProducts + 1),
    LastUpdated = CURRENT_TIMESTAMP
WHERE StatId = 1
RETURNING (SELECT ProductId FROM inserted_product)

CHANGES APPLIED:
- Removed DECLARE statement
- Used CTE (inserted_product) with RETURNING to capture new ProductId
- Chained INSERT operations using CTEs
- Final RETURNING clause returns the ProductId to C# code via ExecuteScalarAsync()

FIX 2: UpdateProductAsync - COMPLETED
===================================================================================

ORIGINAL SQL (SQL Server Syntax - INVALID for PostgreSQL):
----------------------------------------------------------
-- PostgreSQL transaction (use BEGIN or application-level transaction)
    -- Store old values for history
    DECLARE @OldPrice DECIMAL(18,2);
    DECLARE @OldStock INT;
    
    SELECT @OldPrice = Price, @OldStock = StockQuantity
    FROM Products
    WHERE ProductId = @ProductId;
    
    -- Update the product
    UPDATE Products
    SET 
        Name = @Name,
        Description = @Description,
        Price = @Price,
        StockQuantity = @StockQuantity,
        ModifiedDate = CURRENT_TIMESTAMP
    WHERE ProductId = @ProductId;
    
    -- Log the changes
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    VALUES (@ProductId, 'UPDATE', @OldPrice, @Price, @OldStock, @StockQuantity, CURRENT_TIMESTAMP);
    
    -- Update product statistics
    UPDATE ProductStats
    SET 
        AveragePrice = (AveragePrice * TotalProducts - @OldPrice + @Price) / TotalProducts,
        LastUpdated = CURRENT_TIMESTAMP
    WHERE StatId = 1;
-- PostgreSQL COMMIT

PROBLEMS:
- DECLARE @OldPrice and DECLARE @OldStock - Not supported in PostgreSQL
- SELECT @OldPrice = Price - Variable assignment syntax not supported
- VALUES (..., @OldPrice, ..., @OldStock, ...) - Variables not set

FIXED SQL (PostgreSQL CTE with Captured Values):
------------------------------------------------
WITH old_values AS (
    SELECT Price as OldPrice, StockQuantity as OldStock
    FROM Products
    WHERE ProductId = @ProductId
),
product_update AS (
    UPDATE Products
    SET 
        Name = @Name,
        Description = @Description,
        Price = @Price,
        StockQuantity = @StockQuantity,
        ModifiedDate = CURRENT_TIMESTAMP
    WHERE ProductId = @ProductId
    RETURNING ProductId
),
history_insert AS (
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    SELECT @ProductId, 'UPDATE', ov.OldPrice, @Price, ov.OldStock, @StockQuantity, CURRENT_TIMESTAMP
    FROM old_values ov
    RETURNING ProductId
)
UPDATE ProductStats
SET 
    AveragePrice = (AveragePrice * TotalProducts - (SELECT OldPrice FROM old_values) + @Price) / TotalProducts,
    LastUpdated = CURRENT_TIMESTAMP
WHERE StatId = 1

CHANGES APPLIED:
- Removed DECLARE statements
- Used CTE (old_values) to capture old Price and StockQuantity before update
- Chained UPDATE and INSERT operations using CTEs
- Subquery references to old_values CTE for ProductStats calculation

    WHERE StatId = 1;
-- PostgreSQL COMMIT

PROBLEMS:
- DECLARE @OldPrice and DECLARE @OldStock - Not supported in PostgreSQL
- SELECT @OldPrice = Price - Variable assignment syntax not supported
- VALUES (..., @OldPrice, ..., @OldStock, ...) - Variables not set
- CASE expression references @OldPrice - Variable not available

FIXED SQL (PostgreSQL CTE with Pre-Delete Capture):
---------------------------------------------------
WITH old_values AS (
    SELECT Price as OldPrice, StockQuantity as OldStock
    FROM Products
    WHERE ProductId = @ProductId
),
history_insert AS (
    INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
    SELECT @ProductId, 'DELETE', ov.OldPrice, NULL, ov.OldStock, NULL, CURRENT_TIMESTAMP
    FROM old_values ov
    RETURNING ProductId
),
product_delete AS (
    DELETE FROM Products 
    WHERE ProductId = @ProductId
    RETURNING ProductId
)
UPDATE ProductStats
SET 
    TotalProducts = TotalProducts - 1,
    AveragePrice = CASE 
        WHEN TotalProducts > 1 
        THEN (AveragePrice * TotalProducts - (SELECT OldPrice FROM old_values)) / (TotalProducts - 1)
        ELSE 0
    END,
    LastUpdated = CURRENT_TIMESTAMP
WHERE StatId = 1

CHANGES APPLIED:
- Removed DECLARE statements
- Used CTE (old_values) to capture old Price and StockQuantity BEFORE deletion
- Chained INSERT and DELETE operations using CTEs
- Subquery reference to old_values CTE for ProductStats calculation
