public async Task<int> InsertProductAsync(Product product)
{
    var connection = await GetConnectionAsync();
    using var transaction = await connection.BeginTransactionAsync();
    
    try
    {
        // Command 1: Insert and get ID
        const string insertSql = @"
            INSERT INTO Products (Name, Description, Price, StockQuantity)
            VALUES (@Name, @Description, @Price, @StockQuantity)
            RETURNING ProductId";
        
        int newProductId;
        using (var cmd = new NpgsqlCommand(insertSql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@Name", product.Name);
            cmd.Parameters.AddWithValue("@Description", (object)product.Description ?? DBNull.Value);
            cmd.Parameters.AddWithValue("@Price", product.Price);
            cmd.Parameters.AddWithValue("@StockQuantity", product.StockQuantity);
            newProductId = Convert.ToInt32(await cmd.ExecuteScalarAsync());
        }
        
        // Command 2: Log history
        const string historySql = @"
            INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
            VALUES (@ProductId, 'INSERT', NULL, @Price, NULL, @StockQuantity, CURRENT_TIMESTAMP)";
        
        using (var cmd = new NpgsqlCommand(historySql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@ProductId", newProductId);
            cmd.Parameters.AddWithValue("@Price", product.Price);
            cmd.Parameters.AddWithValue("@StockQuantity", product.StockQuantity);
            await cmd.ExecuteNonQueryAsync();
        }
        
        // Command 3: Update stats
        const string statsSql = @"
            UPDATE ProductStats
            SET TotalProducts = TotalProducts + 1,
                AveragePrice = (AveragePrice * TotalProducts + @Price) / (TotalProducts + 1),
                LastUpdated = CURRENT_TIMESTAMP
            WHERE StatId = 1";
        
        using (var cmd = new NpgsqlCommand(statsSql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@Price", product.Price);
            await cmd.ExecuteNonQueryAsync();
        }
        
        await transaction.CommitAsync();
        return newProductId;
    }
    catch
    {
        await transaction.RollbackAsync();
        throw;
    }
}
```

- BEGIN TRANSACTION/COMMIT → Application-level NpgsqlTransaction

public async Task UpdateProductAsync(Product product)
{
    var connection = await GetConnectionAsync();
    using var transaction = await connection.BeginTransactionAsync();
    
    try
    {
        // Command 1: Get old values
        const string getOldSql = "SELECT Price, StockQuantity FROM Products WHERE ProductId = @ProductId";
        decimal oldPrice;
        int oldStock;
        
        using (var cmd = new NpgsqlCommand(getOldSql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@ProductId", product.ProductId);
            using var reader = await cmd.ExecuteReaderAsync();
            if (await reader.ReadAsync())
            {
                oldPrice = reader.GetDecimal(0);
                oldStock = reader.GetInt32(1);
            }
            else
            {
                throw new InvalidOperationException($"Product {product.ProductId} not found");
            }
        }
        
        // Command 2: Update product
        const string updateSql = @"
            UPDATE Products
            SET Name = @Name, Description = @Description, Price = @Price,
                StockQuantity = @StockQuantity, ModifiedDate = CURRENT_TIMESTAMP
            WHERE ProductId = @ProductId";
        
        using (var cmd = new NpgsqlCommand(updateSql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@ProductId", product.ProductId);
            cmd.Parameters.AddWithValue("@Name", product.Name);
            cmd.Parameters.AddWithValue("@Description", (object)product.Description ?? DBNull.Value);
            cmd.Parameters.AddWithValue("@Price", product.Price);
            cmd.Parameters.AddWithValue("@StockQuantity", product.StockQuantity);
            await cmd.ExecuteNonQueryAsync();
        }
        
        // Command 3: Log history
        const string historySql = @"
            INSERT INTO ProductHistory (ProductId, Action, OldPrice, NewPrice, OldStock, NewStock, ActionDate)
            VALUES (@ProductId, 'UPDATE', @OldPrice, @NewPrice, @OldStock, @NewStock, CURRENT_TIMESTAMP)";
        
        using (var cmd = new NpgsqlCommand(historySql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@ProductId", product.ProductId);
            cmd.Parameters.AddWithValue("@OldPrice", oldPrice);
            cmd.Parameters.AddWithValue("@NewPrice", product.Price);
            cmd.Parameters.AddWithValue("@OldStock", oldStock);
            cmd.Parameters.AddWithValue("@NewStock", product.StockQuantity);
            await cmd.ExecuteNonQueryAsync();
        }
        
        // Command 4: Update stats
        const string statsSql = @"
            UPDATE ProductStats
            SET AveragePrice = (AveragePrice * TotalProducts - @OldPrice + @NewPrice) / TotalProducts,
                LastUpdated = CURRENT_TIMESTAMP
            WHERE StatId = 1";
        
        using (var cmd = new NpgsqlCommand(statsSql, connection, transaction))
        {
            cmd.Parameters.AddWithValue("@OldPrice", oldPrice);
            cmd.Parameters.AddWithValue("@NewPrice", product.Price);
            await cmd.ExecuteNonQueryAsync();
        }
        
        await transaction.CommitAsync();
    }
    catch
    {
        await transaction.RollbackAsync();
        throw;
    }
}
```

Key Conversion Points:
- GETDATE() → CURRENT_TIMESTAMP (3 replacements)
- Variable declarations moved to C# code
- SELECT assignment converted to ExecuteReaderAsync()
- Multi-statement batch split into separate commands
- Application-level transaction management
